# 客户端运行
创建：
flutter create .
[Environment]::SetEnvironmentVariable("NO_PROXY", "localhost,127.0.0.1,*.local", "User")

运行：
flutter run -d windows

发行版编译：
```
# Windows 桌面版
flutter build windows
# Android（连上手机或模拟器）
flutter build apk
```


# 远程部署服务器：
cd agent 
.\deploy.bat        

# 本地git:
git add -A
git commit -m ""
